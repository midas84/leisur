select doctor.nombre, paciente.nombres, solicitud.fechacreacion, solicitud.atenciondia, solicitud.diagnostico, analisisespecifico.nombre, solicitud.preciototalestimado from solicitud inner join doctor on solicitud.iddoctor=doctor.id inner join paciente on paciente.id=solicitud.idpaciente inner join atencion on atencion.idsolicitud=solicitud.id inner join analisisespecifico on atencion.idanalisis=analisisespecifico.id
